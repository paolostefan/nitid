3//! # Nitid Transpiler Library
4//!
5//! Exposes the full compilation pipeline so that tests and external
6//! tools can drive it programmatically.
7//!
8//! The exported [`compile`] and [`compile_file`] functions run the
9//! entire four-phase pipeline:
10//!
11//! 1. **Lex**   — [`crate::lexer::Lexer`] tokenizes the source.
12//! 2. **Parse** — [`crate::parser::Parser`] builds an AST.
13//! 3. **Sema**  — [`crate::sema::Sema`] performs semantic analysis.
14//! 4. **Codegen** — [`crate::codegen::Codegen`] emits C source + CMake.
15//!
16//! # Limitations
17//! * Import resolution is declared in the grammar but **not implemented**.
18//! * The semantic analyser does **not** enforce
19//!   no-implicit-casts or no-uninitialized-variables in practice
20//!   (it issues warnings but does not reject the program).
21//! * Memory management, race-condition prevention, and buffer-overflow
22//!   protection are **not** implemented.

44/// Read a file from `path` and transpile it.
45///
46/// This is a convenience wrapper around [`compile`].
47pub fn compile_file(
48    path: &str,
49    c_src_dir: &str,
50) -> Result<(ast::Program, Vec<codegen::CFile>, String), String> {
51    let content = fs::read_to_string(path)
52        .map_err(|e| format!("Failed to read '{}': {}", path, e))?;
53    compile(path, &content, c_src_dir)
54}

56/// Run the full transpilation pipeline on `content`.
57///
58/// # Arguments
59/// * `path` — source file path (used for error messages and C output name).
60/// * `content` — source text of the `.nt` file.
61/// * `c_src_dir` — output directory for generated C files.
62///
63/// # Returns
64/// `(Program, Vec<CFile>, cmake_project_string)` on success,
65/// or a human-readable error message.
66pub fn compile(
67    path: &str,
68    content: &str,
69    c_src_dir: &str,
70) -> Result<(ast::Program, Vec<codegen::CFile>, String), String> {
71    let mut program = parser::Parser::parse(content, path)?;
72    let mut sema_ctx = sema::Sema::new();
73    sema_ctx.analyze(&mut program)?;
74    let mut cg = codegen::Codegen::new();
75    let (c_files, cmake) = cg.generate(&program, c_src_dir)?;
76    Ok((program, c_files, cmake))
77}
