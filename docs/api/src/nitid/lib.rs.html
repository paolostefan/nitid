3//! # Nitid Transpiler Library
4//!
5//! Exposes the full compilation pipeline so that tests and external
6//! tools can drive it programmatically.
7//!
8//! The exported [`compile`] and [`compile_file`] functions run the
9//! entire four-phase pipeline:
10//!
11//! 1. **Lex**   — [`lexer::Lexer`] tokenizes the source.
12//! 2. **Parse** — [`parser::Parser`] builds an AST.
13//! 3. **Sema**  — [`sema::Sema`] performs semantic analysis.
14//! 4. **Codegen** — [`codegen::Codegen`] emits C source + CMake.
15//!
16//! # Limitations
17//! * Import resolution is declared in the grammar but **not implemented**.
18//! * The semantic analyzer does **not** enforce
19//!   no-implicit-casts or no-uninitialized-variables in practice
20//!   (it issues warnings but does not reject the program).
21//! * Memory management, race-condition prevention, and buffer-overflow
22//!   protection are **not** implemented.

52/// Simplified function signature for cross-file resolution.
53#[derive(Debug, Clone)]
54pub struct FunctionSig {
55    pub param_types: Vec<types::Type>,
56    pub return_types: Vec<types::Type>,
57    pub param_names: Vec<String>,
58}

60/// All declarations from a package, flattened across files.
61///
62/// This is the "scope" that Sema uses to resolve imported names.
63/// One PackageContext per imported package.
64#[derive(Debug, Clone)]
65pub struct PackageContext {
66    /// Functions: name -> (param types, return types, param names)
67    pub functions: HashMap<String, FunctionSig>,
68    /// Struct definitions: name -> list of (field name, field type)
69    pub structs: HashMap<String, Vec<(String, types::Type)>>,
70    /// Enum definitions: name -> list of (variant name, optional value)
71    pub enums: HashMap<String, Vec<(String, Option<i128>)>>,
72}

74/// Read a file from `path` and transpile it.
75///
76/// This is a convenience wrapper around [`compile`].
77pub fn compile_file(
78    path: &str,
79    c_src_dir: &str,
80) -> Result<(Program, Vec<codegen::CFile>, String), Diagnostic> {
81    let content = fs::read_to_string(path).map_err(|e| {
82        Diagnostic::new(
83            Phase::Lex,
84            Span::new(path, 0, 0),
85            format!("Failed to read '{}': {}", path, e),
86        )
87    })?;
88    compile(path, &content, c_src_dir)
89}

91/// Read a `.nt` file and parse it into an AST
92///
93/// Runs only the lexer+parser - no semantic analysis or codegen.
94/// This is used to import package files.
95pub fn parse_file(path: &str) -> Result<Program, Diagnostic> {
96    let content = fs::read_to_string(path).map_err(|e| {
97        Diagnostic::new(
98            Phase::Parse,
99            Span::new(path, 0, 0),
100            format!("Failed to read '{}': {}", path, e),
101        )
102    })?;
103    parser::Parser::parse(&content, path)
104}

106/// Parse all `.nt` files in a package dir (recursively).
107///
108/// Returns a Vec of parsed Programs, one per file.
109///Files are parsed in alphabetical order for determinism.
110pub fn parse_package_dir(dir: &Path, expected_package: &str) -> Result<Vec<Program>, Diagnostic> {
111    let mut files = collect_nt_files(dir)?;
112    files.sort(); // deterministic order
113
114    let mut programs = Vec::new();
115    for file_path in &files {
116        let path_str = file_path.to_string_lossy().to_string();
117        let program = parse_file(&path_str)?;
118
119        // Spec rule 3: verify package declaration matches.
120        if program.package != expected_package {
121            return Err(Diagnostic::new(
122                Phase::Import,
123                Span::new(&path_str, 0, 0),
124                format!(
125                    "Package declaration '{}' does not match expected package '{}'",
126                    program.package, expected_package
127                ),
128            ));
129        }
130        eprintln!(
131            "[import] Parsed {}/{}",
132            expected_package,
133            file_path.file_name().unwrap_or_default().to_string_lossy()
134        );
135        programs.push(program);
136    }
137
138    Ok(programs)
139}

141/// Recursively collect all `.nt` files in a directory.
142fn collect_nt_files(dir: &Path) -> Result<Vec<std::path::PathBuf>, Diagnostic> {
143    let mut files = Vec::new();
144    let dir_name = dir.display().to_string();
145    let entries = dir.read_dir().map_err(|e| {
146        Diagnostic::new(
147            Phase::Import,
148            Span::new(&dir_name, 0, 0),
149            format!("Failed to read directory '{}': {}", dir.display(), e),
150        )
151    })?;
152
153    for entry in entries {
154        let entry = entry.map_err(|e| {
155            Diagnostic::new(
156                Phase::Import,
157                Span::new(&dir_name, 0, 0),
158                format!("Failed to read dir entry: {}", e),
159            )
160        })?;
161        let path = entry.path();
162
163        if path.is_dir() {
164            // Recurse into subdirs (subdirectory inheritance assumed)
165            files.extend(collect_nt_files(&path)?);
166        } else if path.extension().map_or(false, |ext| ext == "nt") {
167            files.push(path);
168        }
169    }
170
171    Ok(files)
172}

174/// Resolve an import name to a package directory path.
175///
176/// Given the importing file's path and an import name (e.g. "Math"),
177/// search for a directory named `{name}/` containing `.nt` files.
178///
179/// Search order:
180/// 1. same dir as the importing file
181/// 2. parent dir of the importing file
182/// 3. None (not found)
183fn resolve_package_dir(importing_file: &str, package_name: &str) -> Option<std::path::PathBuf> {
184    let importing_dir = Path::new(importing_file).parent()?;
185
186    // Search 1: sibling directory
187    let candidate = importing_dir.join(package_name);
188    if candidate.is_dir() && has_nt_files(&candidate) {
189        return Some(candidate);
190    }
191
192    // Search 2: parent directory
193    let parent = importing_dir.parent()?;
194    let candidate = parent.join(package_name);
195    if candidate.is_dir() && has_nt_files(&candidate) {
196        return Some(candidate);
197    }
198
199    None
200}

202/// Check if a directory contains any `.nt` files.
203fn has_nt_files(dir: &Path) -> bool {
204    dir.read_dir().ok().map_or(false, |entries| {
205        entries
206            .filter_map(|e| e.ok())
207            .any(|e| e.path().extension().map_or(false, |ext| ext == "nt"))
208    })
209}

211/// Resolve and parse all imports in a program.
212///
213/// For each `import Foo;`, finds the Foo package directory,
214/// parses all `.nt` files inside it, and returns them keyed
215/// by package name.
216pub fn load_imports(program: &Program) -> Result<HashMap<String, Vec<Program>>, Diagnostic> {
217    let mut imported = HashMap::new();
218    let mut visiting = HashSet::new();
219    load_package(program, &mut imported, &mut visiting)?;
220
221    Ok(imported)
222}
223
224fn load_package(
225    program: &Program,
226    imported: &mut HashMap<String, Vec<Program>>,
227    visiting: &mut HashSet<String>,
228) -> Result<(), Diagnostic> {
229    for imp in &program.imports {
230        let key = imp.alias.as_ref().unwrap_or(&imp.name).clone();
231
232        // Already fully loaded (from some other path) => skip
233        if imported.contains_key(&key) {
234            continue;
235        }
236
237        // On the current DFS path => genuine cycle
238        if !visiting.insert(key.clone()) {
239            return Err(Diagnostic::new(
240                Phase::Import,
241                Span::new(&imp.span.file, imp.span.line, imp.span.col),
242                format!("circular import involving '{}'", key),
243            ));
244        }
245
246        let pkg_dir = resolve_package_dir(&program.file, &imp.name).ok_or_else(|| {
247            Diagnostic::new(
248                Phase::Import,
249                Span::new(&imp.span.file, imp.span.line, imp.span.col),
250                format!("could not find package '{}'", imp.name),
251            )
252        })?;
253
254        let programs = parse_package_dir(&pkg_dir, &imp.name)?;
255        imported.insert(key.clone(), programs.clone());
256
257        // Transitive: this package's own imports.
258        for prog in programs {
259            load_package(&prog, imported, visiting)?;
260        }
261
262        visiting.remove(&key);
263    }
264    Ok(())
265}

267fn load_imports_inner(
268    program: &Program,
269    imported: &mut HashMap<String, Vec<Program>>,
270    visited: &mut HashSet<String>,
271) -> Result<HashMap<String, Vec<Program>>, Diagnostic> {
272    for imp in &program.imports {
273        let key = imp.alias.as_ref().unwrap_or(&imp.name).clone();
274
275        if !visited.insert(key.clone()) {
276            return Err(Diagnostic::new(
277                Phase::Import,
278                Span::new(&imp.span.file, imp.span.line, imp.span.col),
279                format!("Circular or duplicate import key '{}'", key),
280            ));
281        }
282
283        if imported.contains_key(&key) {
284            return Err(Diagnostic::new(
285                Phase::Import,
286                Span::new(&imp.span.file, imp.span.line, imp.span.col),
287                format!("Duplicate import key '{}'", key),
288            ));
289        }
290
291        let pkg_dir = resolve_package_dir(&program.file, &imp.name).ok_or_else(|| {
292            Diagnostic::new(
293                Phase::Import,
294                Span::new(&imp.span.file, imp.span.line, imp.span.col),
295                format!(
296                    "Could not find package '{}' (no directory named '{}' found)",
297                    imp.name, imp.name
298                ),
299            )
300        })?;
301
302        let programs = parse_package_dir(&pkg_dir, &imp.name)?;
303        eprintln!(
304            "[import] Loaded package '{}' ({} files)",
305            imp.name,
306            programs.len()
307        );
308
309        // import alias (if any) or package name
310        imported.insert(key, programs);
311    }
312
313    Ok(imported.clone())
314}

316/// Build a PackageContext from a list of parsed programs (one package)
317///
318/// Extracts all function, struct and enum declarations from every file in the package into a flat,
319/// unified view.
320pub fn build_package_context(programs: &[Program]) -> PackageContext {
321    let mut ctx = PackageContext {
322        functions: HashMap::new(),
323        structs: HashMap::new(),
324        enums: HashMap::new(),
325    };
326
327    for program in programs {
328        for decl in &program.decls {
329            match decl {
330                ast::Decl::FnDecl(f) => {
331                    let param_types: Vec<types::Type> = f
332                        .params
333                        .iter()
334                        .flat_map(|p| std::iter::repeat(p.typ.clone()).take(p.names.len()))
335                        .collect();
336                    let param_names: Vec<String> =
337                        f.params.iter().flat_map(|p| p.names.clone()).collect();
338                    ctx.functions.insert(
339                        f.name.clone(),
340                        FunctionSig {
341                            param_types,
342                            return_types: f.returns.clone(),
343                            param_names,
344                        },
345                    );
346                }
347                ast::Decl::StructDecl(s) => {
348                    let fields: Vec<(String, types::Type)> = s
349                        .fields
350                        .iter()
351                        .map(|f| (f.name.clone(), f.typ.clone()))
352                        .collect();
353                    ctx.structs.insert(s.name.clone(), fields);
354                }
355                ast::Decl::EnumDecl(e) => {
356                    let variants: Vec<(String, Option<i128>)> = e
357                        .variants
358                        .iter()
359                        .map(|v| {
360                            let val = match &v.value {
361                                Some(ast::Expr::IntLit(n, _)) => Some(*n),
362                                _ => None,
363                            };
364                            (v.name.clone(), val)
365                        })
366                        .collect();
367                    ctx.enums.insert(e.name.clone(), variants);
368                }
369                // ImplBlock methods are handled via struct_methods in sema.as
370                _ => {}
371            }
372        }
373    }
374    ctx
375}

377/// Combine multiple PackageContexts into one.
378///
379/// Used when a file imports several packages. Later contexts shadow earlier ones
380/// on name conflicts (last-one-wins).
381pub fn merge_contexts(contexts: &[PackageContext]) -> PackageContext {
382    let mut merged = PackageContext {
383        functions: HashMap::new(),
384        structs: HashMap::new(),
385        enums: HashMap::new(),
386    };
387
388    for ctx in contexts {
389        merged
390            .functions
391            .extend(ctx.functions.iter().map(|(k, v)| (k.clone(), v.clone())));
392        merged
393            .structs
394            .extend(ctx.structs.iter().map(|(k, v)| (k.clone(), v.clone())));
395        merged
396            .enums
397            .extend(ctx.enums.iter().map(|(k, v)| (k.clone(), v.clone())));
398    }
399
400    merged
401}

403/// Run the full transpilation pipeline on `content`.
404///
405/// # Arguments
406/// * `path` — source file path (used for error messages and C output name).
407/// * `content` — source text of the `.nt` file.
408/// * `c_src_dir` — output directory for generated C files.
409///
410/// # Returns
411/// `(Program, Vec<CFile>, cmake_project_string)` on success,
412/// or a human-readable error message.
413pub fn compile(
414    path: &str,
415    content: &str,
416    c_src_dir: &str,
417) -> Result<(Program, Vec<codegen::CFile>, String), Diagnostic> {
418    let mut program = parser::Parser::parse(content, path)?;
419
420    // Load and parse imports.
421    let imports = load_imports(&program)?;
422
423    // Build a per-package context map.
424    let pkg_contexts: HashMap<String, PackageContext> = imports
425        .iter()
426        .map(|(name, programs)| (name.clone(), build_package_context(programs)))
427        .collect();
428
429    // Every function exported by every imported package.
430    let mut foreign_sigs: HashMap<String, (String, Vec<types::Type>, Vec<types::Type>)> =
431        HashMap::new();
432    for (pkg_name, ctx) in &pkg_contexts {
433        for (name, sig) in &ctx.functions {
434            // Mangle: "multiply" in package "Math" -> "Math_multiply"
435            let mangled = format!("{}_{}", pkg_name, name);
436            foreign_sigs.insert(
437                name.clone(),
438                (mangled, sig.param_types.clone(), sig.return_types.clone()),
439            );
440        }
441    }
442
443    // Pass the package map to sema.
444    let mut sema_ctx = sema::Sema::new();
445    sema_ctx.analyze(&mut program, Some(&pkg_contexts))?;
446
447    let mut cg = codegen::Codegen::new();
448    let package_names: HashSet<String> = pkg_contexts.keys().cloned().collect();
449    let (mut c_files, _) = cg
450        .generate(&program, c_src_dir, package_names.clone(), &foreign_sigs)
451        .expect("codegen does not fail");
452
453    // Emit a .c file for every file in every imported package.
454    for (pkg_name, programs) in &imports {
455        // Prevent self-functions from appearing as "foreign"
456        let pkg_foreign = foreign_sigs
457            .iter()
458            .filter(|(_, (mangled, _, _))| !mangled.starts_with(&format!("{}_", pkg_name)))
459            .map(|(k, v)| (k.clone(), v.clone()))
460            .collect();
461
462        for prog in programs {
463            let base_name = Path::new(&prog.file)
464                .file_stem()
465                .and_then(|s| s.to_str())
466                .unwrap_or("pkg");
467            let c_text = cg.generate_c(prog, package_names.clone(), &pkg_foreign);
468            c_files.push(codegen::CFile {
469                path: format!("{}/{}.c", c_src_dir, base_name),
470                content: c_text,
471            });
472        }
473    }
474
475    // Stems for add_executable, derived from every emitted .c.
476    let stems: Vec<String> = c_files
477        .iter()
478        .map(|cf| {
479            Path::new(&cf.path)
480                .file_stem()
481                .unwrap_or_default()
482                .to_string_lossy()
483                .into_owned()
484        })
485        .collect();
486    let cmake = cg.emit_cmake(&program, &stems);
487
488    Ok((program, c_files, cmake))
489}
