1/// Semantic analyser for Nitid.
2///
3/// Phase 3 of the transpilation pipeline: validates the AST produced
4/// by the parser and annotates it with type information.
5///
6/// # Checks performed
7/// * Variable scoping (declare / lookup in nested scopes).
8/// * Return-value count matches the function signature.
9/// * Binary operator type compatibility (both operands must have the
10///   same type for arithmetic / bitwise ops).
11/// * Function call arity and existence.
12///
13/// # Checks NOT yet performed (TODO)
14/// * No implicit type casts → **enforced**: literal values are
15///   range-checked against the declared numeric type.
16/// * Uninitialised variables → **not enforced**: only a warning is
17///   issued for `:=` declarations with inferred types.
18/// * Overflow / underflow protection.
19/// * Dead code detection.
20use std::collections::HashMap;
21use crate::ast::*;
22use crate::types::{is_string_type, Type};
23
24/// Info about a struct definition: its fields and layout attributes.
25#[derive(Debug, Clone)]
26struct StructInfo {
27    fields: Vec<StructField>,
28    packed: bool,
29    align: Option<u64>,
30}
31
32/// Info about an enum definition: its underlying type and variants.
33#[derive(Debug, Clone)]
34struct EnumInfo {
35    underlying_type: Type,
36    variants: Vec<EnumVariant>,
37}
38
39/// The semantic analyser context.
40///
41/// Holds the global scope, a reference to the current function's
42/// return types, and accumulates warnings during analysis.
43pub struct Sema {
44    pub warnings: Vec<String>,
45    pub errors: Vec<String>,
46    globals: Scope,
47    current_fn_returns: Vec<Type>,
48    multi_return_temps: HashMap<String, Vec<String>>,
49    /// Stack of uninitialized variable sets. Each level tracks vars
50    /// declared (without init) in that scope. Variables are removed
51    /// when assigned. At scope exit, remaining vars produce an error.
52    uninit_scopes: Vec<HashMap<String, Span>>,
53    /// Current loop nesting depth. break/continue only valid when > 0.
54    loop_depth: usize,
55    /// Collected struct definitions.
56    struct_defs: HashMap<String, StructInfo>,
57    /// Methods per struct: struct_name -> Vec<(method_name, FnDecl)>.
58    struct_methods: HashMap<String, Vec<FnDecl>>,
59    /// Whether we are currently analyzing a method body (for self injection).
60    current_struct_type: Option<Type>,
61    /// Collected enum definitions.
62    enum_defs: HashMap<String, EnumInfo>,
63    /// Enum member lookup: member_name -> (underlying_type, value).
64    enum_members: HashMap<String, (Type, i128)>,
65}
66
67/// A lexical scope mapping variable names to their types.
68///
69/// Scopes form a chain: each scope has an optional parent.  Variable
70/// lookup walks up the chain.
71struct Scope {
72    vars: HashMap<String, Type>,
73    parent: Option<Box<Scope>>,
74}
75
76impl Scope {
77    /// Create an empty scope with no parent.
78    fn new() -> Self {
79        Self { vars: HashMap::new(), parent: None }
80    }
81
82    /// Create a child scope with `parent` as the enclosing scope.
83    fn child(parent: Scope) -> Self {
84        Self { vars: HashMap::new(), parent: Some(Box::new(parent)) }
85    }
86
87    /// Declare a variable in *this* scope only.
88    fn declare(&mut self, name: &str, typ: Type, span: &Span) -> Result<(), String> {
89        if self.vars.contains_key(name) {
90            return Err(format!("{}:{}:{}: Variable '{}' already declared in this scope",
91                span.file, span.line, span.col, name));
92        }
93        self.vars.insert(name.to_string(), typ);
94        Ok(())
95    }
96
97    /// Look up a variable, checking this scope first, then parents.
98    fn lookup(&self, name: &str) -> Option<Type> {
99        self.vars.get(name).cloned().or_else(|| {
100            self.parent.as_ref().and_then(|p| p.lookup(name))
101        })
102    }
103}

131    /// Analyse all top-level declarations.
132    ///
133    /// Pre-pass: collect struct definitions.
134    /// Pass 1: collect function & method signatures for call resolution.
135    /// Pass 2: analyse each declaration body.
136    fn analyze_decls(&mut self, decls: &mut [Decl]) -> Result<(), String> {
137        // Pre-pass: collect struct and enum definitions.
138        for decl in decls.iter() {
139            match decl {
140                Decl::StructDecl(s) => {
141                    let name = s.name.clone();
142                    if self.struct_defs.contains_key(&name) {
143                        return Err(format!(
144                            "{}:{}:{}: Duplicate struct definition '{}'",
145                            s.span.file, s.span.line, s.span.col, name
146                        ));
147                    }
148                    self.struct_defs.insert(name, StructInfo { fields: s.fields.clone(), packed: s.packed, align: s.align });
149                }
150                Decl::EnumDecl(e) => {
151                    let name = e.name.clone();
152                    if self.enum_defs.contains_key(&name) {
153                        return Err(format!(
154                            "{}:{}:{}: Duplicate enum definition '{}'",
155                            e.span.file, e.span.line, e.span.col, name
156                        ));
157                    }
158                    // Determine underlying type (default i32).
159                    let underlying = e.typ.clone().unwrap_or(Type::I32);
160                    // Only integral types allowed.
161                    if !is_integral_type(&underlying) {
162                        return Err(format!(
163                            "{}:{}:{}: Enum '{}' must use an integral type, got '{}'",
164                            e.span.file, e.span.line, e.span.col, name, underlying
165                        ));
166                    }
167                    // Check for duplicate variant names and evaluate values.
168                    let mut seen = std::collections::HashSet::new();
169                    let mut next_val: i128 = 0;
170                    for v in &e.variants {
171                        if !seen.insert(v.name.clone()) {
172                            return Err(format!(
173                                "{}:{}:{}: Duplicate variant '{}' in enum '{}'",
174                                v.span.file, v.span.line, v.span.col, v.name, name
175                            ));
176                        }
177                        let val = if let Some(ref expr) = v.value {
178                            eval_enum_value(expr, &v.span)?
179                        } else {
180                            next_val
181                        };
182                        // Check value fits in the underlying type.
183                        if !int_lit_fits(val, &underlying) {
184                            return Err(format!(
185                                "{}:{}:{}: Value {} does not fit in enum '{}' type '{}'",
186                                v.span.file, v.span.line, v.span.col, val, name, underlying
187                            ));
188                        }
189                        self.enum_members.insert(v.name.clone(), (Type::Enum(name.clone()), val));
190                        next_val = val.checked_add(1).ok_or_else(|| {
191                            format!("{}:{}:{}: Enum '{}' variant values overflow",
192                                v.span.file, v.span.line, v.span.col, name)
193                        })?;
194                    }
195                    self.enum_defs.insert(name, EnumInfo { underlying_type: underlying, variants: e.variants.clone() });
196                }
197                _ => {}
198            }
199        }
200
201        // Pass 1: gather all function signatures and method signatures.
202        let mut fn_sigs: HashMap<String, (Vec<Type>, Vec<Type>)> = HashMap::new();
203        for decl in decls.iter() {
204            match decl {
205                Decl::FnDecl(f) => {
206                    let param_types: Vec<Type> = f.params.iter().flat_map(|p| {
207                        std::iter::repeat(p.typ.clone()).take(p.names.len())
208                    }).collect();
209                    fn_sigs.insert(f.name.clone(), (param_types, f.returns.clone()));
210                }
211                Decl::ImplBlock(imp) => {
212                    let struct_name = &imp.struct_name;
213                    if !self.struct_defs.contains_key(struct_name) {
214                        return Err(format!(
215                            "{}:{}:{}: Cannot define methods for unknown struct '{}'",
216                            imp.span.file, imp.span.line, imp.span.col, struct_name
217                        ));
218                    }
219                    for method in &imp.methods {
220                        let m_name = method.name.clone();
221                        // Check for duplicate method names within this impl block
222                        // (will be caught by the struct_methods insert below)
223                        let param_types: Vec<Type> = method.params.iter().flat_map(|p| {
224                            std::iter::repeat(p.typ.clone()).take(p.names.len())
225                        }).collect();
226                        // Register method in fn_sigs with mangled name
227                        let mangled = format!("{}_{}", struct_name, m_name);
228                        // Method gets an implicit self pointer parameter in C
229                        let mut c_params = vec![Type::Struct(struct_name.clone())];
230                        c_params.extend(param_types);
231                        fn_sigs.insert(mangled, (c_params, method.returns.clone()));
232                    }
233                    self.struct_methods.insert(struct_name.clone(), imp.methods.clone());
234                }
235                _ => {}
236            }
237        }
238
239        // Pass 2: analyse bodies.
240        for decl in decls.iter_mut() {
241            match decl {
242                Decl::FnDecl(f) => {
243                    self.current_fn_returns = f.returns.clone();
244                    self.current_struct_type = None;
245                    let mut scope = Scope::new();
246                    for param in &f.params {
247                        for name in &param.names {
248                            scope.declare(name, param.typ.clone(), &param.span)?;
249                        }
250                    }
251                    self.analyze_block(&mut f.body, &mut scope, &fn_sigs)?;
252                }
253                Decl::ImplBlock(imp) => {
254                    let struct_name = imp.struct_name.clone();
255                    let struct_type = Type::Struct(struct_name.clone());
256                    for method in &mut imp.methods {
257                        self.current_fn_returns = method.returns.clone();
258                        self.current_struct_type = Some(struct_type.clone());
259                        let mut scope = Scope::new();
260                        // Declare self as the struct type
261                        let self_span = method.span.clone();
262                        scope.declare("self", struct_type.clone(), &self_span)?;
263                        for param in &method.params {
264                            for name in &param.names {
265                                scope.declare(name, param.typ.clone(), &param.span)?;
266                            }
267                        }
268                        self.analyze_block(&mut method.body, &mut scope, &fn_sigs)?;
269                    }
270                    self.current_struct_type = None;
271                }
272                Decl::VarDecl(v) => {
273                    analyze_var_decl_free(v, &mut self.globals, &fn_sigs, &mut self.warnings, &self.struct_defs, &self.enum_members)?;
274                    let can_zero_init = v.typ.as_ref().map(|t| matches!(t, Type::TyArray(..) | Type::TyFixedArray(..))).unwrap_or(false);
275                    if v.init.is_none() && !can_zero_init {
276                        for name in &v.names {
277                            return Err(format!(
278                                "{}:{}:{}: Variable '{}' must be initialized",
279                                v.span.file, v.span.line, v.span.col, name
280                            ));
281                        }
282                    }
283                }
284                Decl::StructDecl(_) | Decl::EnumDecl(_) => {} // already handled
285            }
286        }
287        Ok(())
288    }

292    /// Analyse a block of statements, creating a nested scope.
293    fn analyze_block(
294        &mut self,
295        stmts: &mut [Stmt],
296        scope: &mut Scope,
297        fn_sigs: &HashMap<String, (Vec<Type>, Vec<Type>)>,
298    ) -> Result<(), String> {
299        self.push_uninit_scope();
300        let mut local_scope = Scope::child(std::mem::replace(scope, Scope::new()));
301        for stmt in stmts.iter_mut() {
302            self.analyze_stmt(stmt, &mut local_scope, fn_sigs)?;
303        }
304        *scope = *local_scope.parent.unwrap();
305        self.pop_uninit_scope()
306    }

308    /// Analyse a single statement.
309    fn analyze_stmt(
310        &mut self,
311        stmt: &mut Stmt,
312        scope: &mut Scope,
313        fn_sigs: &HashMap<String, (Vec<Type>, Vec<Type>)>,
314    ) -> Result<(), String> {
315        match stmt {
316            Stmt::VarDecl(v) => {
317                analyze_var_decl_free(v, scope, fn_sigs, &mut self.warnings, &self.struct_defs, &self.enum_members)?;
318                // Deferred init check: vars without init are tracked
319                // per scope and checked at scope exit.
320                // Array types can be zero-initialized so skip uninit tracking.
321                let can_zero_init = v.typ.as_ref().map(|t| matches!(t, Type::TyArray(..) | Type::TyFixedArray(..))).unwrap_or(false);
322                if v.init.is_none() && !can_zero_init {
323                    for name in &v.names {
324                        self.add_uninit(name, &v.span);
325                    }
326                }
327                Ok(())
328            }
329            Stmt::Expr(e) => {
330                self.analyze_expr(e, scope, fn_sigs)?;
331                Ok(())
332            }
333            Stmt::Return(values, span) => {
334                // Check that the number of returned values matches the
335                // function signature.
336                if values.len() != self.current_fn_returns.len() {
337                    return Err(format!(
338                        "{}:{}:{}: Return {} values but function expects {}",
339                        span.file, span.line, span.col,
340                        values.len(), self.current_fn_returns.len()
341                    ));
342                }
343                for val in values.iter_mut() {
344                    self.analyze_expr(val, scope, fn_sigs)?;
345                }
346                Ok(())
347            }
348            Stmt::Block(body) => {
349                self.push_uninit_scope();
350                let mut inner = Scope::child(std::mem::replace(scope, Scope::new()));
351                for stmt in body.iter_mut() {
352                    self.analyze_stmt(stmt, &mut inner, fn_sigs)?;
353                }
354                *scope = *inner.parent.unwrap();
355                self.pop_uninit_scope()
356            }
357            Stmt::If { cond, then_block, else_block, .. } => {
358                self.analyze_expr(cond, scope, fn_sigs)?;
359                self.analyze_block(then_block, scope, fn_sigs)?;
360                if let Some(else_b) = else_block {
361                    self.analyze_block(else_b, scope, fn_sigs)?;
362                }
363                Ok(())
364            }
365            Stmt::While { cond, body, .. } => {
366                self.analyze_expr(cond, scope, fn_sigs)?;
367                self.loop_depth += 1;
368                self.analyze_block(body, scope, fn_sigs)?;
369                self.loop_depth -= 1;
370                Ok(())
371            }
372            Stmt::For { init, cond, inc, body, .. } => {
373                if let Some(init_stmt) = init {
374                    self.analyze_stmt(init_stmt, scope, fn_sigs)?;
375                }
376                if let Some(cond_expr) = cond {
377                    self.analyze_expr(cond_expr, scope, fn_sigs)?;
378                }
379                if let Some(inc_expr) = inc {
380                    self.analyze_expr(inc_expr, scope, fn_sigs)?;
381                }
382                self.loop_depth += 1;
383                self.analyze_block(body, scope, fn_sigs)?;
384                self.loop_depth -= 1;
385                Ok(())
386            }
387            Stmt::ForIn { var, iter, body, span } => {
388                let iter_type = self.analyze_expr(iter, scope, fn_sigs)?;
389                if !matches!(iter_type, Type::TyArray(..) | Type::TyFixedArray(..)) {
390                    return Err(format!(
391                        "{}:{}:{}: Cannot iterate over non-array type",
392                        span.file, span.line, span.col
393                    ));
394                }
395                self.push_uninit_scope();
396                let mut inner = Scope::child(std::mem::replace(scope, Scope::new()));
397                inner.declare(var, Type::I32, span)?;
398                self.loop_depth += 1;
399                for stmt in body.iter_mut() {
400                    self.analyze_stmt(stmt, &mut inner, fn_sigs)?;
401                }
402                self.loop_depth -= 1;
403                *scope = *inner.parent.unwrap();
404                self.pop_uninit_scope()
405            }
406            Stmt::ForInIndex { idx_var, item_var, iter, body, span } => {
407                let iter_type = self.analyze_expr(iter, scope, fn_sigs)?;
408                if !matches!(iter_type, Type::TyArray(..) | Type::TyFixedArray(..)) {
409                    return Err(format!(
410                        "{}:{}:{}: Cannot iterate over non-array type",
411                        span.file, span.line, span.col
412                    ));
413                }
414                self.push_uninit_scope();
415                let mut inner = Scope::child(std::mem::replace(scope, Scope::new()));
416                inner.declare(idx_var, Type::I32, span)?;
417                inner.declare(item_var, Type::I32, span)?;
418                self.loop_depth += 1;
419                for stmt in body.iter_mut() {
420                    self.analyze_stmt(stmt, &mut inner, fn_sigs)?;
421                }
422                self.loop_depth -= 1;
423                *scope = *inner.parent.unwrap();
424                self.pop_uninit_scope()
425            }
426            Stmt::Break(span) => {
427                if self.loop_depth == 0 {
428                    return Err(format!(
429                        "{}:{}:{}: 'break' outside of loop",
430                        span.file, span.line, span.col
431                    ));
432                }
433                Ok(())
434            }
435            Stmt::Continue(span) => {
436                if self.loop_depth == 0 {
437                    return Err(format!(
438                        "{}:{}:{}: 'continue' outside of loop",
439                        span.file, span.line, span.col
440                    ));
441                }
442                Ok(())
443            }
444        }
445    }

449    fn analyze_expr(
450        &mut self,
451        expr: &mut Expr,
452        scope: &mut Scope,
453        fn_sigs: &HashMap<String, (Vec<Type>, Vec<Type>)>,
454    ) -> Result<Type, String> {
455        let typ = self.infer_expr_type(expr, scope, fn_sigs)?;
456        // Track assignments to initialize variables.
457        if let Expr::Assign { left, .. } = expr {
458            if let Expr::Ident(name, _) = left.as_ref() {
459                self.mark_initialized(name);
460            }
461        }
462        if let Expr::PostIncrement { target, .. } | Expr::PostDecrement { target, .. } = expr {
463            if let Expr::Ident(name, _) = target.as_ref() {
464                self.mark_initialized(name);
465            }
466        }
467        Ok(typ)
468    }

470    /// Infer the type of an expression.
471    ///
472    /// This is the core type-checking function.  It recursively walks
473    /// the expression tree and:
474    /// - Assigns fixed types to literals (int → `I32`, float → `F64`, etc.).
475    /// - Resolves identifiers against the scope chain.
476    /// - Validates function call arguments and looks up the return type.
477    /// - Checks binary-operator type compatibility.
478    fn infer_expr_type(
479        &self,
480        expr: &Expr,
481        scope: &Scope,
482        fn_sigs: &HashMap<String, (Vec<Type>, Vec<Type>)>,
483    ) -> Result<Type, String> {
484        match expr {
485            Expr::IntLit(..) => Ok(Type::I32),
486            Expr::FloatLit(..) => Ok(Type::F64),
487            Expr::StringLit(..) => Ok(Type::String),
488            Expr::CharLit(..) => Ok(Type::U8),
489            Expr::BoolLit(..) => Ok(Type::Bool),
490            Expr::Ident(name, span) => {
491                scope.lookup(name).or_else(|| {
492                    self.enum_members.get(name).map(|(t, _)| t.clone())
493                }).ok_or_else(|| {
494                    format!("{}:{}:{}: Undefined variable '{}'", span.file, span.line, span.col, name)
495                })
496            }
497            Expr::Call { name, args, span } => {
498                for arg in args {
499                    self.infer_expr_type(arg, scope, fn_sigs)?;
500                }
501                // Built-in functions.
502                if name == "println" || name == "printf" {
503                    Ok(Type::Void)
504                } else if let Some(typ) = Type::from_str(name) {
505                    // Type conversion: `i64(expr)` returns the target type
506                    Ok(typ)
507                } else {
508                    fn_sigs.get(name).map(|(_, returns)| {
509                        if returns.len() == 1 {
510                            returns[0].clone()
511                        } else {
512                            // Multi-return functions have void type in expression
513                            // context; the return values are written through
514                            // output pointer arguments.
515                            Type::Void
516                        }
517                    }).ok_or_else(|| format!("{}:{}:{}: Undefined function '{}'",
518                        span.file, span.line, span.col, name))
519                }
520            }
521            Expr::BinaryOp { left, right, op, span } => {
522                let lt = self.infer_expr_type(left, scope, fn_sigs)?;
523                let rt = self.infer_expr_type(right, scope, fn_sigs)?;
524                let lt_r = resolve_enum_type(&lt, &self.enum_defs).clone();
525                let rt_r = resolve_enum_type(&rt, &self.enum_defs).clone();
526                match op {
527                    BinOp::Add | BinOp::Sub | BinOp::Mul | BinOp::Div | BinOp::Mod => {
528                        if lt_r == rt_r { Ok(lt_r) } else {
529                            Err(format!("{}:{}:{}: Type mismatch in arithmetic",
530                                span.file, span.line, span.col))
531                        }
532                    }
533                    BinOp::Eq | BinOp::Ne | BinOp::Lt | BinOp::Gt | BinOp::Le | BinOp::Ge => {
534                        Ok(Type::Bool)
535                    }
536                    BinOp::And | BinOp::Or => Ok(Type::Bool),
537                    BinOp::BitAnd | BinOp::BitOr | BinOp::BitXor | BinOp::Shl | BinOp::Shr => {
538                        if lt_r == rt_r { Ok(lt_r) } else {
539                            Err(format!("{}:{}:{}: Type mismatch in bitwise op",
540                                span.file, span.line, span.col))
541                        }
542                    }
543                }
544            }
545            Expr::Assign { left, .. } => self.infer_expr_type(left, scope, fn_sigs),
546            Expr::DeclAssign { right, .. } => self.infer_expr_type(right, scope, fn_sigs),
547            Expr::ArrayLit(elems, span) => {
548                if elems.is_empty() {
549                    return Err(format!("{}:{}:{}: Empty array literals not supported yet",
550                        span.file, span.line, span.col));
551                }
552                let elem_type = self.infer_expr_type(&elems[0], scope, fn_sigs)?;
553                for e in elems[1..].iter() {
554                    let et = self.infer_expr_type(e, scope, fn_sigs)?;
555                    if et != elem_type {
556                        return Err(format!("{}:{}:{}: Array element type mismatch",
557                            span.file, span.line, span.col));
558                    }
559                }
560                Ok(Type::TyArray(Box::new(elem_type), None))
561            }
562            Expr::Index { target, index, span } => {
563                let target_type = self.infer_expr_type(target, scope, fn_sigs)?;
564                let index_type = self.infer_expr_type(index, scope, fn_sigs)?;
565                if !is_numeric_type(&index_type) {
566                    return Err(format!("{}:{}:{}: Array index must be numeric",
567                        span.file, span.line, span.col));
568                }
569                match target_type {
570                    // Fixed-size arrays: check compile-time known index bounds.
571                    // Negative indices wrap from end (-1 → last element, -n → index 0)
572                    Type::TyFixedArray(elem, sz) => check_index_bounds(&elem, Some(sz), index, span),
573                    // Dynamically-sized arrays can be resized at runtime, so
574                    // the declared size is not a compile-time bound; illegal
575                    // accesses are caught at runtime instead.
576                    Type::TyArray(elem, _) => Ok(*elem),
577                    Type::String | Type::String16 | Type::String32 => Ok(Type::U32),
578                    _ => Err(format!("{}:{}:{}: Cannot index non-array type",
579                        span.file, span.line, span.col)),
580                }
581            }
582            Expr::PostIncrement { target, span } | Expr::PostDecrement { target, span } => {
583                let t = self.infer_expr_type(target, scope, fn_sigs)?;
584                if !is_numeric_type(&t) {
585                    return Err(format!(
586                        "{}:{}:{}: Cannot apply ++ or -- to non-numeric type {:?}",
587                        span.file, span.line, span.col, t
588                    ));
589                }
590                Ok(t)
591            }
592            Expr::FieldAccess { target, field, span } => {
593                let target_type = self.infer_expr_type(target, scope, fn_sigs)?;
594                match target_type {
595                    Type::Struct(ref struct_name) => {
596                        let info = self.struct_defs.get(struct_name).ok_or_else(|| {
597                            format!("{}:{}:{}: Unknown struct type '{}'",
598                                span.file, span.line, span.col, struct_name)
599                        })?;
600                        info.fields.iter().find(|f| f.name == *field)
601                            .map(|f| f.typ.clone())
602                            .ok_or_else(|| {
603                                format!("{}:{}:{}: Struct '{}' has no field '{}'",
604                                    span.file, span.line, span.col, struct_name, field)
605                            })
606                    }
607                    _ => Err(format!("{}:{}:{}: Cannot access field on non-struct type",
608                        span.file, span.line, span.col)),
609                }
610            }
611            Expr::MethodCall { target, method, args, span } => {
612                let target_type = self.infer_expr_type(target, scope, fn_sigs)?;
613                match target_type {
614                    Type::Struct(ref struct_name) => {
615                        let methods = self.struct_methods.get(struct_name).ok_or_else(|| {
616                            format!("{}:{}:{}: Struct '{}' has no methods",
617                                span.file, span.line, span.col, struct_name)
618                        })?;
619                        let method_decl = methods.iter().find(|m| m.name == *method)
620                            .ok_or_else(|| {
621                                format!("{}:{}:{}: Struct '{}' has no method '{}'",
622                                    span.file, span.line, span.col, struct_name, method)
623                            })?;
624                        // Check method arity
625                        let expected_params: Vec<&Type> = method_decl.params.iter()
626                            .flat_map(|p| std::iter::repeat(&p.typ).take(p.names.len()))
627                            .collect();
628                        if args.len() != expected_params.len() {
629                            return Err(format!(
630                                "{}:{}:{}: Method '{}' expects {} arguments, got {}",
631                                span.file, span.line, span.col, method,
632                                expected_params.len(), args.len()
633                            ));
634                        }
635                        // Check argument types (for now just validate they exist)
636                        for arg in args {
637                            self.infer_expr_type(arg, scope, fn_sigs)?;
638                        }
639                        // Return the method's return type
640                        if method_decl.returns.len() == 1 {
641                            Ok(method_decl.returns[0].clone())
642                        } else if method_decl.returns.is_empty() || method_decl.returns[0] == Type::Void {
643                            Ok(Type::Void)
644                        } else {
645                            Ok(Type::Void) // multi-return in expression context
646                        }
647                    }
648                    Type::TyArray(..) | Type::TyFixedArray(..) => check_array_method(method, args, span, &target_type,
649                        |e| self.infer_expr_type(e, scope, fn_sigs)),
650                    _ => Err(format!("{}:{}:{}: Cannot call method on non-struct type",
651                        span.file, span.line, span.col)),
652                }
653            }
654            Expr::StructLit { struct_name, fields, span } => {
655                let info = self.struct_defs.get(struct_name).ok_or_else(|| {
656                    format!("{}:{}:{}: Unknown struct type '{}'",
657                        span.file, span.line, span.col, struct_name)
658                })?;
659                // Check field count matches
660                if fields.len() != info.fields.len() {
661                    return Err(format!(
662                        "{}:{}:{}: Struct '{}' has {} fields but literal provides {}",
663                        span.file, span.line, span.col, struct_name,
664                        info.fields.len(), fields.len()
665                    ));
666                }
667                // Check each field exists and validate types
668                for (field_name, field_expr) in fields {
669                    let decl_field = info.fields.iter().find(|f| f.name == *field_name)
670                        .ok_or_else(|| {
671                            format!("{}:{}:{}: Struct '{}' has no field '{}'",
672                                span.file, span.line, span.col, struct_name, field_name)
673                        })?;
674                    let init_type = self.infer_expr_type(field_expr, scope, fn_sigs)?;
675                    if init_type != decl_field.typ {
676                        let ok = is_numeric_type(&init_type) && is_numeric_type(&decl_field.typ)
677                            && literal_fits_target(field_expr, &decl_field.typ);
678                        let ok = ok || {
679                            // Array of numerics: check each element fits target element type.
680                            if let (Some((init_elem, _)), Some((decl_elem, _))) = (as_array(&init_type), as_array(&decl_field.typ)) {
681                                if is_numeric_type(init_elem) && is_numeric_type(decl_elem) && init_elem != decl_elem {
682                                    if let Expr::ArrayLit(elems, _) = field_expr {
683                                        elems.iter().all(|e| literal_fits_target(e, decl_elem))
684                                    } else {
685                                        false
686                                    }
687                                } else {
688                                    init_elem == decl_elem
689                                }
690                            } else {
691                                false
692                            }
693                        };
694                        if !ok {
695                            let sp = field_expr.span();
696                            return Err(format!("{}:{}:{}: Type mismatch for field '{}': expected {:?}, got {:?}",
697                                sp.file, sp.line, sp.col, field_name, decl_field.typ, init_type));
698                        }
699                    }
700                }
701                Ok(Type::Struct(struct_name.clone()))
702            }
703        }
704    }

706    // ── Uninitialized variable tracking ─────────────────────
707
708    /// Push a new scope for uninitialized variable tracking.
709    fn push_uninit_scope(&mut self) {
710        self.uninit_scopes.push(HashMap::new());
711    }
712
713    /// Pop the current uninit scope; error if any vars remain uninitialized.
714    fn pop_uninit_scope(&mut self) -> Result<(), String> {
715        if let Some(scope) = self.uninit_scopes.pop() {
716            for (name, span) in &scope {
717                return Err(format!(
718                    "{}:{}:{}: Variable '{}' must be initialized",
719                    span.file, span.line, span.col, name
720                ));
721            }
722        }
723        Ok(())
724    }
725
726    /// Record `name` as declared-but-uninitialized in the current scope.
727    fn add_uninit(&mut self, name: &str, span: &Span) {
728        if let Some(scope) = self.uninit_scopes.last_mut() {
729            scope.insert(name.to_string(), span.clone());
730        }
731    }
732
733    /// Remove `name` from all uninit scopes (innermost first) —
734    /// called when an assignment to `name` is found.
735    fn mark_initialized(&mut self, name: &str) {
736        for scope in self.uninit_scopes.iter_mut().rev() {
737            scope.remove(name);
738        }
739    }
740}

744/// Analyse a variable declaration, inferring types and checking
745/// initialiser compatibility.
746///
747/// This function is called both for global and local variable declarations.
748fn analyze_var_decl_free(
749    v: &mut VarDecl,
750    scope: &mut Scope,
751    fn_sigs: &HashMap<String, (Vec<Type>, Vec<Type>)>,
752    warnings: &mut Vec<String>,
753    struct_defs: &HashMap<String, StructInfo>,
754    enum_members: &HashMap<String, (Type, i128)>,
755) -> Result<(), String> {
756    let inferred_type = if let Some(ref init) = v.init {
757        Some(infer_expr_type_free(init, scope, fn_sigs, struct_defs, enum_members)?)
758    } else {
759        None
760    };
761
762    for name in &v.names {
763        let typ = if let Some(ref t) = v.typ {
764            t.clone()
765        } else if let Some(ref t) = inferred_type {
766            warnings.push(format!(
767                "{}:{}:{}: Variable '{}' declared with ':=', inferred type {:?}",
768                v.span.file, v.span.line, v.span.col, name, t
769            ));
770            t.clone()
771        } else {
772            return Err(format!("{}:{}:{}: Variable '{}' has no type and no initializer",
773                v.span.file, v.span.line, v.span.col, name));
774        };
775
776        // Type-check the initializer against the declared type.
777        if let Some(ref init) = v.init {
778            let init_type = infer_expr_type_free(init, scope, fn_sigs, struct_defs, enum_members)?;
779            // Resolve enum types to underlying type for comparison.
780            let init_r = match &init_type { Type::Enum(_) => &Type::I32, _ => &init_type };
781            let typ_r = match &typ { Type::Enum(_) => &Type::I32, _ => &typ };
782            if init_r != typ_r {
783                if is_numeric_type(init_r) && is_numeric_type(typ_r) {
784                    // Numeric-to-numeric: check if the literal value fits.
785                    if !literal_fits_target(init, &typ) {
786                        return Err(format!(
787                            "{}:{}:{}: Value does not fit in '{}' of type {:?}",
788                            v.span.file, v.span.line, v.span.col, name, typ
789                        ));
790                    }
791                } else if let (Some((init_elem, _)), Some((decl_elem, _))) = (as_array(&init_type), as_array(&typ)) {
792                    // Array of numerics: check each element literal fits target element type.
793                    if is_numeric_type(init_elem) && is_numeric_type(decl_elem) && init_elem != decl_elem {
794                        if let Expr::ArrayLit(elems, _) = init {
795                            for elem in elems {
796                                if !literal_fits_target(elem, decl_elem) {
797                                    let sp = elem.span();
798                                    return Err(format!(
799                                        "{}:{}:{}: Value does not fit in array element type {:?}",
800                                        sp.file, sp.line, sp.col, decl_elem
801                                    ));
802                                }
803                            }
804                        }
805                    } else if init_elem != decl_elem && typ != Type::Void {
806                        return Err(format!(
807                            "{}:{}:{}: Type mismatch: '{}' is {:?}, initializer is {:?}",
808                            v.span.file, v.span.line, v.span.col, name, typ, init_type
809                        ));
810                    }
811                } else if is_string_type(init_r) && is_string_type(typ_r) {
812                    // String literals / string-typed values can be assigned to any string type
813                } else if typ != Type::Void {
814                    return Err(format!(
815                        "{}:{}:{}: Type mismatch: '{}' is {:?}, initializer is {:?}",
816                        v.span.file, v.span.line, v.span.col, name, typ, init_type
817                    ));
818                }
819            }
820        }
821
822        scope.declare(name, typ, &v.span)?;
823    }
824    Ok(())
825}

827/// Infer the type of an expression without a surrounding `Sema` context.
828///
829/// Duplicates the logic in `Sema::infer_expr_type` but is a standalone
830/// function so it can be called from `analyze_var_decl_free` without
831/// borrowing issues.
832/// Check a method call on an array-typed target.
833///
834/// `infer_arg` infers the type of an argument expression; it is supplied
835/// by the caller because array methods are validated from two different
836/// inference contexts (`Sema::infer_expr_type` and the free-standing
837/// `infer_expr_type_free`).
838fn check_array_method<F>(
839    method: &str,
840    args: &[Expr],
841    span: &Span,
842    array_type: &Type,
843    mut infer_arg: F,
844) -> Result<Type, String>
845where
846    F: FnMut(&Expr) -> Result<Type, String>,
847{
848    match method {
849        "size" => {
850            if !args.is_empty() {
851                return Err(format!(
852                    "{}:{}:{}: Array method 'size' expects 0 arguments, got {}",
853                    span.file, span.line, span.col, args.len()
854                ));
855            }
856            Ok(Type::I32)
857        }
858        "resize" => {
859            // Only dynamically-sized arrays can be resized; fixed-size
860            // arrays (`fixed` keyword) are backed by plain C arrays.
861            if matches!(array_type, Type::TyFixedArray(..)) {
862                return Err(format!(
863                    "{}:{}:{}: Cannot resize fixed-size array",
864                    span.file, span.line, span.col
865                ));
866            }
867            if args.len() != 1 {
868                return Err(format!(
869                    "{}:{}:{}: Array method 'resize' expects 1 argument, got {}",
870                    span.file, span.line, span.col, args.len()
871                ));
872            }
873            let arg_type = infer_arg(&args[0])?;
874            if !is_integral_type(&arg_type) {
875                return Err(format!(
876                    "{}:{}:{}: Array method 'resize' expects an integer argument, got {}",
877                    span.file, span.line, span.col, arg_type
878                ));
879            }
880            // A negative literal can never be a valid length. Unary minus
881            // is desugared to `0 - lit`, which extract_index_literal folds.
882            if let Some(val) = extract_index_literal(&args[0]) {
883                if val < 0 {
884                    return Err(format!(
885                        "{}:{}:{}: Array method 'resize' expects a non-negative size, got {}",
886                        span.file, span.line, span.col, val
887                    ));
888                }
889            }
890            Ok(Type::Void)
891        }
892        _ => Err(format!("{}:{}:{}: Array type has no method '{}'",
893            span.file, span.line, span.col, method)),
894    }
895}

897fn infer_expr_type_free(
898    expr: &Expr,
899    scope: &Scope,
900    fn_sigs: &HashMap<String, (Vec<Type>, Vec<Type>)>,
901    struct_defs: &HashMap<String, StructInfo>,
902    enum_members: &HashMap<String, (Type, i128)>,
903) -> Result<Type, String> {
904    match expr {
905        Expr::IntLit(..) => Ok(Type::I32),
906        Expr::FloatLit(..) => Ok(Type::F64),
907        Expr::StringLit(..) => Ok(Type::String),
908        Expr::CharLit(..) => Ok(Type::U8),
909        Expr::BoolLit(..) => Ok(Type::Bool),
910        Expr::Ident(name, span) => {
911            scope.lookup(name).or_else(|| {
912                enum_members.get(name).map(|(t, _)| t.clone())
913            }).ok_or_else(|| {
914                format!("{}:{}:{}: Undefined variable '{}'", span.file, span.line, span.col, name)
915            })
916        }
917        Expr::Call { name, args, span } => {
918            for arg in args {
919                infer_expr_type_free(arg, scope, fn_sigs, struct_defs, enum_members)?;
920            }
921            if name == "println" || name == "printf" {
922                Ok(Type::Void)
923            } else if let Some(typ) = Type::from_str(name) {
924                Ok(typ)
925            } else {
926                fn_sigs.get(name).map(|(_, returns)| {
927                    if returns.len() == 1 {
928                        returns[0].clone()
929                    } else {
930                        Type::Void
931                    }
932                }).ok_or_else(|| format!("{}:{}:{}: Undefined function '{}'",
933                    span.file, span.line, span.col, name))
934            }
935        }
936        Expr::BinaryOp { left, right, op, span } => {
937            let lt = infer_expr_type_free(left, scope, fn_sigs, struct_defs, enum_members)?;
938            let rt = infer_expr_type_free(right, scope, fn_sigs, struct_defs, enum_members)?;
939            // Resolve enum types to their underlying type for comparison.
940            let lt_r = match &lt { Type::Enum(_) => Type::I32, _ => lt.clone() };
941            let rt_r = match &rt { Type::Enum(_) => Type::I32, _ => rt.clone() };
942            match op {
943                BinOp::Add | BinOp::Sub | BinOp::Mul | BinOp::Div | BinOp::Mod => {
944                    if lt_r == rt_r { Ok(lt_r) } else {
945                        Err(format!("{}:{}:{}: Type mismatch in arithmetic",
946                            span.file, span.line, span.col))
947                    }
948                }
949                BinOp::Eq | BinOp::Ne | BinOp::Lt | BinOp::Gt | BinOp::Le | BinOp::Ge => {
950                    Ok(Type::Bool)
951                }
952                BinOp::And | BinOp::Or => Ok(Type::Bool),
953                BinOp::BitAnd | BinOp::BitOr | BinOp::BitXor | BinOp::Shl | BinOp::Shr => {
954                    if lt_r == rt_r { Ok(lt_r) } else {
955                        Err(format!("{}:{}:{}: Type mismatch in bitwise op",
956                            span.file, span.line, span.col))
957                    }
958                }
959            }
960        }
961        Expr::Assign { left, .. } => infer_expr_type_free(left, scope, fn_sigs, struct_defs, enum_members),
962        Expr::DeclAssign { right, .. } => infer_expr_type_free(right, scope, fn_sigs, struct_defs, enum_members),
963        Expr::ArrayLit(elems, span) => {
964            if elems.is_empty() {
965                return Err(format!("{}:{}:{}: Empty array literals not supported yet",
966                    span.file, span.line, span.col));
967            }
968            let elem_type = infer_expr_type_free(&elems[0], scope, fn_sigs, struct_defs, enum_members)?;
969            for e in elems[1..].iter() {
970                let et = infer_expr_type_free(e, scope, fn_sigs, struct_defs, enum_members)?;
971                if et != elem_type {
972                    return Err(format!("{}:{}:{}: Array element type mismatch",
973                        span.file, span.line, span.col));
974                }
975            }
976            Ok(Type::TyArray(Box::new(elem_type), None))
977        }
978        Expr::Index { target, index, span } => {
979            let target_type = infer_expr_type_free(target, scope, fn_sigs, struct_defs, enum_members)?;
980            let index_type = infer_expr_type_free(index, scope, fn_sigs, struct_defs, enum_members)?;
981            if !is_numeric_type(&index_type) {
982                return Err(format!("{}:{}:{}: Array index must be numeric",
983                    span.file, span.line, span.col));
984            }
985            match target_type {
986                // Fixed-size arrays: check compile-time known index bounds.
987                // Negative indices wrap from end (-1 → last element, -n → index 0)
988                Type::TyFixedArray(elem, sz) => check_index_bounds(&elem, Some(sz), index, span),
989                // Dynamically-sized arrays can be resized at runtime, so
990                // the declared size is not a compile-time bound; illegal
991                // accesses are caught at runtime instead.
992                Type::TyArray(elem, _) => Ok(*elem),
993                Type::String | Type::String16 | Type::String32 => Ok(Type::U32),
994                _ => Err(format!("{}:{}:{}: Cannot index non-array type",
995                    span.file, span.line, span.col)),
996            }
997        }
998        Expr::PostIncrement { target, .. } | Expr::PostDecrement { target, .. } => {
999            infer_expr_type_free(target, scope, fn_sigs, struct_defs, enum_members)
1000        }
1001        Expr::FieldAccess { target, field, span } => {
1002            let target_type = infer_expr_type_free(target, scope, fn_sigs, struct_defs, enum_members)?;
1003            match target_type {
1004                Type::Struct(ref struct_name) => {
1005                    let info = struct_defs.get(struct_name).ok_or_else(|| {
1006                        format!("{}:{}:{}: Unknown struct type '{}'",
1007                            span.file, span.line, span.col, struct_name)
1008                    })?;
1009                    info.fields.iter().find(|f| f.name == *field)
1010                        .map(|f| f.typ.clone())
1011                        .ok_or_else(|| {
1012                            format!("{}:{}:{}: Struct '{}' has no field '{}'",
1013                                span.file, span.line, span.col, struct_name, field)
1014                        })
1015                }
1016                _ => Err(format!("{}:{}:{}: Cannot access field on non-struct type",
1017                    span.file, span.line, span.col)),
1018            }
1019        }
1020        Expr::MethodCall { target, method, args, span } => {
1021            let target_type = infer_expr_type_free(target, scope, fn_sigs, struct_defs, enum_members)?;
1022            match target_type {
1023                Type::Struct(_) => {
1024                    for arg in args {
1025                        infer_expr_type_free(arg, scope, fn_sigs, struct_defs, enum_members)?;
1026                    }
1027                    Ok(Type::Void)
1028                }
1029                Type::TyArray(..) | Type::TyFixedArray(..) => check_array_method(method, args, span, &target_type,
1030                    |e| infer_expr_type_free(e, scope, fn_sigs, struct_defs, enum_members)),
1031                _ => Err(format!("{}:{}:{}: Cannot call method on non-struct type",
1032                    span.file, span.line, span.col)),
1033            }
1034        }
1035        Expr::StructLit { struct_name, fields, span } => {
1036            let info = struct_defs.get(struct_name).ok_or_else(|| {
1037                format!("{}:{}:{}: Unknown struct type '{}'",
1038                    span.file, span.line, span.col, struct_name)
1039            })?;
1040            if fields.len() != info.fields.len() {
1041                return Err(format!(
1042                    "{}:{}:{}: Struct '{}' has {} fields but literal provides {}",
1043                    span.file, span.line, span.col, struct_name,
1044                    info.fields.len(), fields.len()
1045                ));
1046            }
1047            for (field_name, field_expr) in fields {
1048                let decl_field = info.fields.iter().find(|f| f.name == *field_name)
1049                    .ok_or_else(|| {
1050                        format!("{}:{}:{}: Struct '{}' has no field '{}'",
1051                            span.file, span.line, span.col, struct_name, field_name)
1052                    })?;
1053                let init_type = infer_expr_type_free(field_expr, scope, fn_sigs, struct_defs, enum_members)?;
1054                if init_type != decl_field.typ {
1055                    let ok = is_numeric_type(&init_type) && is_numeric_type(&decl_field.typ)
1056                        && literal_fits_target(field_expr, &decl_field.typ);
1057                    let ok = ok || {
1058                        if let (Some((init_elem, _)), Some((decl_elem, _))) = (as_array(&init_type), as_array(&decl_field.typ)) {
1059                            if is_numeric_type(init_elem) && is_numeric_type(decl_elem) && init_elem != decl_elem {
1060                                if let Expr::ArrayLit(elems, _) = field_expr {
1061                                    elems.iter().all(|e| literal_fits_target(e, decl_elem))
1062                                } else {
1063                                    false
1064                                }
1065                            } else {
1066                                init_elem == decl_elem
1067                            }
1068                        } else {
1069                            false
1070                        }
1071                    };
1072                    if !ok {
1073                        let sp = field_expr.span();
1074                        return Err(format!("{}:{}:{}: Type mismatch for field '{}': expected {:?}, got {:?}",
1075                            sp.file, sp.line, sp.col, field_name, decl_field.typ, init_type));
1076                    }
1077                }
1078            }
1079            Ok(Type::Struct(struct_name.clone()))
1080        }
1081    }
1082}

1084/// Evaluate an enum variant value expression.
1085///
1086/// Handles integer literals and negated integer literals (from `-1` desugaring).
1087fn eval_enum_value(expr: &Expr, span: &Span) -> Result<i128, String> {
1088    match expr {
1089        Expr::IntLit(val, _) => Ok(*val),
1090        Expr::BinaryOp { left, op: BinOp::Sub, right, .. } => {
1091            if let Expr::IntLit(0, _) = left.as_ref() {
1092                if let Expr::IntLit(rval, _) = right.as_ref() {
1093                    return rval.checked_neg().ok_or_else(|| {
1094                        format!("{}:{}:{}: Enum value overflow",
1095                            span.file, span.line, span.col)
1096                    });
1097                }
1098            }
1099            Err(format!("{}:{}:{}: Enum variant value must be an integer literal",
1100                span.file, span.line, span.col))
1101        }
1102        _ => Err(format!("{}:{}:{}: Enum variant value must be an integer literal",
1103            span.file, span.line, span.col)),
1104    }
1105}

1107/// Extract a literal integer from an index expression, handling the
1108/// parser's unary-minus desugaring (`-n` → `0 - n` → `-n`).
1109///
1110/// Returns `None` if the index is not a compile-time known literal.
1111fn extract_index_literal(index: &Expr) -> Option<i128> {
1112    match index {
1113        Expr::IntLit(val, _) => Some(*val),
1114        Expr::BinaryOp { left, op: BinOp::Sub, right, .. } => {
1115            if let Expr::IntLit(0, _) = left.as_ref() {
1116                if let Expr::IntLit(val, _) = right.as_ref() {
1117                    return Some(-val);
1118                }
1119            }
1120            None
1121        }
1122        _ => None,
1123    }
1124}

1126/// Decompose an array type into its element type and optional declared
1127/// size. Covers both dynamically-sized (`TyArray`) and fixed-size
1128/// (`TyFixedArray`) arrays.
1129fn as_array(t: &Type) -> Option<(&Type, Option<u64>)> {
1130    match t {
1131        Type::TyArray(elem, size) => Some((elem, *size)),
1132        Type::TyFixedArray(elem, n) => Some((elem, Some(*n))),
1133        _ => None,
1134    }
1135}

1137/// Check compile-time bounds for a literal index against a declared
1138/// array size. Negative indices wrap from the end (-1 → last element).
1139fn check_index_bounds(elem: &Type, size: Option<u64>, index: &Expr, span: &Span) -> Result<Type, String> {
1140    if let Some(sz) = size {
1141        if let Some(val) = extract_index_literal(index) {
1142            if val < 0 {
1143                let pos = (-val) as u64;
1144                if pos > sz {
1145                    return Err(format!("{}:{}:{}: Array index {} out of bounds for size {}",
1146                        span.file, span.line, span.col, val, sz));
1147                }
1148            } else if (val as u64) >= sz {
1149                return Err(format!("{}:{}:{}: Array index {} out of bounds for size {}",
1150                    span.file, span.line, span.col, val, sz));
1151            }
1152        }
1153    }
1154    Ok(elem.clone())
1155}

1160        Type::I8 | Type::I16 | Type::I32 | Type::I64 | Type::I128 | Type::I256

1162        | Type::F8 | Type::F16 | Type::F32 | Type::F64

1174/// Resolve an enum type to its underlying type; pass through other types unchanged.
1175fn resolve_enum_type<'a>(t: &'a Type, enum_defs: &'a HashMap<String, EnumInfo>) -> &'a Type {
1176    match t {
1177        Type::Enum(name) => enum_defs.get(name).map(|e| &e.underlying_type).unwrap_or(t),
1178        _ => t,
1179    }

1182/// Check if a literal (or simple negated literal) expression's value
1183/// can be assigned to `target` without overflow.
1184fn literal_fits_target(expr: &Expr, target: &Type) -> bool {
1185    // Resolve enum type to underlying type.
1186    let target = match target {
1187        Type::Enum(_) => &Type::I32,
1188        _ => target,
1189    };
1190    match expr {
1191        Expr::IntLit(val, _) => int_lit_fits(*val, target),
1192        Expr::FloatLit(val, _) => float_lit_fits(*val, target),
1193        Expr::BinaryOp { left, op: BinOp::Sub, right, .. } => {
1194            // Detect `0 - lit` (unary minus desugared by the parser).
1195            if let Expr::IntLit(0, _) = left.as_ref() {
1196                if let Expr::IntLit(rval, _) = right.as_ref() {
1197                    return int_lit_fits(-rval, target);
1198                }
1199            }
1200            true
1201        }
1202        _ => true,
1203    }
1204}

1206/// Check if an integer literal value fits in the target type.
1207fn int_lit_fits(val: i128, target: &Type) -> bool {
1208    match target {
1209        Type::I8 => val >= i8::MIN as i128 && val <= i8::MAX as i128,
1210        Type::I16 => val >= i16::MIN as i128 && val <= i16::MAX as i128,
1211        Type::I32 => val >= i32::MIN as i128 && val <= i32::MAX as i128,
1212        Type::I64 => val >= i64::MIN as i128 && val <= i64::MAX as i128,
1213        Type::I128 => true,
1214        Type::I256 => true,
1215        Type::U8 => val >= 0 && val <= u8::MAX as i128,
1216        Type::U16 => val >= 0 && val <= u16::MAX as i128,
1217        Type::U32 => val >= 0 && val <= u32::MAX as i128,
1218        Type::U64 => val >= 0 && val <= u64::MAX as i128,
1219        Type::U128 => val >= 0,
1220        Type::U256 => val >= 0,
1221        Type::F8 | Type::F16 | Type::F32 | Type::F64 => true,
1222        _ => false,
1223    }
1224}

1226/// Check if a float literal value fits in the target type.
1227fn float_lit_fits(_val: f64, target: &Type) -> bool {
1228    match target {
1229        Type::I8 | Type::I16 | Type::I32 | Type::I64 | Type::I128 | Type::I256
1230        | Type::U8 | Type::U16 | Type::U32 | Type::U64 | Type::U128 | Type::U256 => {
1231            // Float literal does not implicitly narrow to integer.
1232            false
1233        }
1234        Type::F8 | Type::F16 | Type::F32 | Type::F64 => true,
1235        _ => false,
1236    }
1237}
